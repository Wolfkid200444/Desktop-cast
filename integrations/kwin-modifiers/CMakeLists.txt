cmake_minimum_required(VERSION 3.20)
project(DesktopCastModifiers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Gui Qml Quick Widgets)
find_package(KWin REQUIRED)
add_library(desktopcast_modifier_bridge MODULE bridge.cpp)
target_link_libraries(desktopcast_modifier_bridge PRIVATE KWin::kwin Qt6::Qml Qt6::DBus)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/kwin-version.txt" "${KWin_VERSION}")
